// دالة تأخير تنفيذ الأوامر
const delay = (ms) => new Promise(resolve => setTimeout(resolve, ms));

// تخزين خيارات المستخدم المؤقتة
const userState = {};

// 1️⃣ القائمة الرئيسية لبدء البوت
bot.onText(/\/start/, (msg) => {
  sendMainMenu(msg.chat.id);
});

function sendMainMenu(chatId) {
  const opts = {
    reply_markup: {
      inline_keyboard: [
        [
          { text: '📊 عرض مخزون الأدوات بالحسابات', callback_data: 'view_inventory' }
        ],
        [
          { text: '🥊 تنزيل قفاز', callback_data: 'select_item_glove' },
          { text: '🧲 تنزيل مغناطيس', callback_data: 'select_item_magnet' }
        ],
        [
          { text: '🧴 تنزيل بخاخ', callback_data: 'select_item_spray' },
          { text: '⏱️ تنزيل وقت', callback_data: 'select_item_time' }
        ]
      ]
    }
  };
  bot.sendMessage(chatId, '🎮 **لوحة التحكم بالداعمات لجاكو**\nاختر العملية التي تريد تنفيذها من الأزرار:', { parse_mode: 'Markdown', ...opts });
}

// 2️⃣ الاستجابة لأزرار البوت
bot.on('callback_query', async (query) => {
  const chatId = query.message.chat.id;
  const data = query.data;

  // عرض المخزون لكل حساب
  if (data === 'view_inventory') {
    let text = '📦 **مخزون الأدوات الحالي في جميع الحسابات:**\n\n';
    accountsList.forEach(acc => {
      text += `👤 **${acc.username}**\n`;
      text += `├ 🥊 قفاز: ${acc.glove}\n`;
      text += `├ 🧲 مغناطيس: ${acc.magnet}\n`;
      text += `├ 🧴 بخاخ: ${acc.spray}\n`;
      text += `└ ⏱️ وقت: ${acc.time}\n\n`;
    });
    
    bot.sendMessage(chatId, text, { parse_mode: 'Markdown' });
    bot.answerCallbackQuery(query.id);
  }

  // اختيار الأداة
  else if (data.startsWith('select_item_')) {
    const item = data.replace('select_item_', '');
    userState[chatId] = { item: item };

    // إنشاء أزرار اختيار الحسابات
    const accountButtons = accountsList.map(acc => [{ text: `👤 ${acc.username}`, callback_data: `select_acc_${acc.username}` }]);
    accountButtons.unshift([{ text: '🌐 جميع الحسابات الـ 5', callback_data: 'select_acc_all' }]);

    bot.sendMessage(chatId, `🎯 اختر الحساب المُنَفِّذ لتنزيل **[${getItemName(item)}]**:`, {
      reply_markup: { inline_keyboard: accountButtons },
      parse_mode: 'Markdown'
    });
    bot.answerCallbackQuery(query.id);
  }

  // اختيار الحساب
  else if (data.startsWith('select_acc_')) {
    const accTarget = data.replace('select_acc_', '');
    if (userState[chatId]) {
      userState[chatId].target = accTarget;
    }

    // خيارات التوقيت والتأخير الزمني
    const delayOpts = {
      reply_markup: {
        inline_keyboard: [
          [{ text: '⚡ تنزيل فوراً (0 ثانية)', callback_data: 'exec_delay_0' }],
          [{ text: '⏱️ بعد 2 ثانية', callback_data: 'exec_delay_2' }],
          [{ text: '⏱️ بعد 5 ثواني', callback_data: 'exec_delay_5' }],
          [{ text: '⏱️ بعد 10 ثواني', callback_data: 'exec_delay_10' }]
        ]
      }
    };
    bot.sendMessage(chatId, `⏳ اختر التوقيت أو التأخير الزمني قبل التنزيل:`, delayOpts);
    bot.answerCallbackQuery(query.id);
  }

  // التنفيذ الفعلي وخصم الأدوات
  else if (data.startsWith('exec_delay_')) {
    const delaySec = parseInt(data.replace('exec_delay_', ''));
    const state = userState[chatId] || { item: 'glove', target: 'all' };
    const itemName = getItemName(state.item);

    bot.sendMessage(chatId, `🚀 جاري تحضير طلب تنزيل **[${itemName}]**...\n👤 الحساب: \`${state.target}\`\n⏳ التأخير: \`${delaySec}\` ثواني`, { parse_mode: 'Markdown' });
    
    if (delaySec > 0) {
      await delay(delaySec * 1000);
    }

    // تحديد الحسابات المستهدفة
    let targets = accountsList;
    if (state.target !== 'all') {
      targets = accountsList.filter(acc => acc.username === state.target);
    }

    let successCount = 0;
    for (let acc of targets) {
      if (acc[state.item] > 0) {
        acc[state.item] -= 1; // خصم الأداة من الحساب
        successCount++;
        bot.sendMessage(chatId, `✅ **[${acc.username}]** تم تنزيل ${itemName} بنجاح!\nالمتبقي بالحساب: \`${acc[state.item]}\``, { parse_mode: 'Markdown' });
      } else {
        bot.sendMessage(chatId, `⚠️ **[${acc.username}]** لا يملك رصيد كافي من ${itemName}!`, { parse_mode: 'Markdown' });
      }
      await delay(2000); // فاصل ثانيتين بين كل حساب لتفادي الباند
    }

    bot.sendMessage(chatId, `🎉 **اكتملت العملية!** تم التنزيل بواسطة (${successCount}) حساب.`, { parse_mode: 'Markdown' });
    bot.answerCallbackQuery(query.id);
  }
});

function getItemName(key) {
  const names = { glove: '🥊 قفاز', magnet: '🧲 مغناطيس', spray: '🧴 بخاخ', time: '⏱️ وقت' };
  return names[key] || key;
}
